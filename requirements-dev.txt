# Development and CI dependencies.
#
# Install with:  pip install -r requirements-dev.txt
#
# This file is intentionally NOT a superset of requirements.txt. The test
# suite exercises ticker_resolver.py and generic_query_builder.py in
# isolation, so it needs neither TA-Lib (which requires a C toolchain) nor
# streamlit, jugaad-data or nselib. Keeping them out is what lets CI run in
# well under a minute on a clean runner.
#
# Major versions are pinned; patch releases are allowed.

# --- Test runner ---------------------------------------------------------
# The suite is verified against both 8.x and 9.x; the pin keeps CI on a single
# major so a new release cannot change collection behaviour unannounced.
pytest>=8.0,<9.0
pytest-cov>=5.0,<7.0

# --- Linting -------------------------------------------------------------
ruff>=0.6,<1.0

# --- Runtime imports pulled in by the modules under test -----------------
# ticker_resolver.py imports pandas at module scope (for the optional CF-CA
# CSV loader). Already a runtime dependency in requirements.txt; repeated here
# so CI can install this file alone.
pandas>=3.0.5,<4.0

# App/config.py imports python-dotenv when available. Installing it means the
# .env code path is the one under test rather than the ImportError fallback.
python-dotenv>=1.0,<2.0

# --- Security scanning (also run as a non-blocking CI job) ---------------
pip-audit>=2.7,<3.0
